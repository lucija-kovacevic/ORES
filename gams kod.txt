* Variables
Positive Variables
    gen(i,b,t);

Variable total_cost;

* Equations
Equations
    obj, balance(t), cap_limit(i,b,t);

obj..
    total_cost =e= sum((i,b,t), gen(i,b,t) * price(i,b,t));

balance(t)..
    sum((i,b), gen(i,b,t)) =e= demand(t);

cap_limit(i,b,t)..
    gen(i,b,t) =l= capacity(i,b,t);

Model market /all/;
Solve market using LP minimizing total_cost;

* === REZULTATI: ANALIZA PO ELEKTRANI I PO SATU ===

Parameters
    gen_sum(i,t)         "Ukupna proizvodnja po elektrani i satu"
    mcp(t)               "Marginalni trošak (dual balans jednadžbe)"
    revenue(i,t)         "Prihod po satu"
    operational_cost(i,t) "Operativni trošak po satu"
    total_cost_i(i)      "Ukupan operativni trošak"
    total_rev_i(i)       "Ukupan prihod"
    total_profit(i)      "Ukupan profit"
    energy_total(i)      "Ukupna prodana energija po elektrani"
    hourly_summary(t,*)  "Vrijeme, A–D, demand, marginalni trošak";

* Sumiraj proizvodnju po elektrani (suma kroz sve bidove)
gen_sum(i,t) = sum(b, gen.l(i,b,t));

* MCP = dual balans jednadžbe
mcp(t) = balance.m(t);

* Prihodi i troškovi po satu
revenue(i,t)         = mcp(t) * gen_sum(i,t);
operational_cost(i,t)= cost(i,t) * gen_sum(i,t); 

* Ukupni rezultati po elektrani
total_rev_i(i)     = sum(t, revenue(i,t));
total_cost_i(i)    = sum(t, operational_cost(i,t));
total_profit(i)    = total_rev_i(i) - total_cost_i(i);
energy_total(i)    = sum(t, gen_sum(i,t));

* === PO SATU: sheet za analizu po satu ===
hourly_summary(t,'Hour')     = ord(t);
hourly_summary(t,'A_MWh')    = gen_sum('A',t);
hourly_summary(t,'B_MWh')    = gen_sum('B',t);
hourly_summary(t,'C_MWh')    = gen_sum('C',t);
hourly_summary(t,'D_MWh')    = gen_sum('D',t);
hourly_summary(t,'Demand')   = demand(t);
hourly_summary(t,'MCP_EUR')  = mcp(t);

display total_profit, energy_total, total_rev_i, total_cost_i;
